function renderBreakdownLists(pada) { // ============ ස්කන්ධ ============ var sList = document.getElementById('skandha-list'); sList.innerHTML = ''; var sCount = 0; pada.skandha.forEach(function(s) { if (s.value && s.value !== 'නැත' && s.value !== '-' && s.value !== '×') sCount++; var div = document.createElement('div'); div.className = 'breakdown-item'; div.innerHTML = '' + s.num + '.' + s.name + '' + s.value + ''; sList.appendChild(div); }); document.getElementById('skandha-count').innerText = sCount + ' / 5'; // ============ ආයතන ============ var aList = document.getElementById('ayatana-list'); aList.innerHTML = ''; var aCount = 0; pada.ayatana.forEach(function(a) { if (a.value && a.value !== 'නැත' && a.value !== '-' && a.value !== '×') aCount++; var div = document.createElement('div'); div.className = 'breakdown-item'; div.innerHTML = '' + a.num + '.' + a.name + '' + a.value + ''; aList.appendChild(div); }); document.getElementById('ayatana-count').innerText = aCount + ' / 12'; // ============ ධාතු ============ var dList = document.getElementById('dhatu-list'); dList.innerHTML = ''; var dCount = 0; pada.dhatu.forEach(function(d) { if (d.value && d.value !== 'නැත' && d.value !== '-' && d.value !== '×') dCount++; var div = document.createElement('div'); div.className = 'breakdown-item'; div.innerHTML = '' + d.num + '.' + d.name + '' + d.value + ''; dList.appendChild(div); }); document.getElementById('dhatu-count').innerText = dCount + ' / 18'; // ============ සත්‍ය ============ var satList = document.getElementById('sathya-list'); satList.innerHTML = ''; var satCount = 0; pada.sathya.forEach(function(sat) { if (sat.value && sat.value !== 'නැත' && sat.value !== '-' && sat.value !== '×' && sat.value !== 'සතය විනිර්මුක්ත') satCount++; var div = document.createElement('div'); div.className = 'breakdown-item'; div.innerHTML = '' + sat.num + '.' + sat.name + '' + sat.value + ''; satList.appendChild(div); }); document.getElementById('sathya-count').innerText = satCount + ' / 4'; }
